// SKETCH (typecheck-only) for migration (a): the haze head-overflow tail of
// src/lib/gpu/look/haze-graph.ts (prepGraphFor / prepGraph, header lines 37-43) with GPU-resident
// control state and a GPU-gated, GPU-sized dispatch instead of deciding everything on the CPU.
//
// What it does, in graph terms (appended after "gather", before the read node):
//   hz-ctl      (1 thread)  starts[LISTS], head → ctl = [total, overflow, dispatchX, 1, 1]
//   tail-pack   (GPU indirect condition on ctl @ byte 8; x written on the GPU = ⌈min(total-head, cap)/64⌉)
//               packs list slots [head, head + min(total-head, cap)) into a transient `tail` window
//   read "head" counts, state, starts, ctl, list heads, sky, and the tail window (one submit)
// CPU after the read: total = ctl[0]; total ≤ head + cap → no second read (the common overflow case);
// total > head + cap → the existing exact readBack of [head + cap, total) (residual round trip).
//
// FINDING this sketch demonstrates: a GPU condition gates *dispatches* only. WebGPU copy sizes
// (copyBufferToBuffer, i.e. every readback) are fixed on the CPU at encode time, and MAP_READ buffers
// cannot be STORAGE, so the GPU cannot size a readback. The tail window costs 2·cap·4 bytes of copy
// whether or not the gate fires: functionally this equals `head += cap`. The gate saves only the
// pack dispatch. See notes/upstream-api.md §7a for the alternative that removes the round trip
// ("copy capacity, map exact").

/** ctl words: 0 total, 1 overflow (total > head), 2..4 the tail-pack indirect dispatch (x, y, z). */
const CTL_WORDS = 5;
const CTL_DISPATCH_BYTE = 8;
const WG = 64;

/** Run-time scalars the graph reads (one uniform: head and cap change per run, not per graph). */
export type TailParams = { head: number; cap: number };

const HZ_CTL = /* wgsl */ `
struct TP { head: u32, cap: u32, _a: u32, _b: u32 };
@group(0) @binding(0) var<uniform> tp: TP;
@group(0) @binding(1) var<storage, read> starts: array<u32>;
@group(0) @binding(2) var<storage, read_write> ctl: array<u32>;
@compute @workgroup_size(1)
fn main() {
	let total = starts[${LISTS}u];
	let over = total > tp.head;
	let n = select(0u, min(total - tp.head, tp.cap), over);
	ctl[0] = total;
	ctl[1] = select(0u, 1u, over);
	ctl[2] = (n + ${WG - 1}u) / ${WG}u;   // 0 workgroups when nothing overflowed: the gate
	ctl[3] = 1u;
	ctl[4] = 1u;
}`;

const HZ_TAIL_PACK = /* wgsl */ `
struct TP { head: u32, cap: u32, _a: u32, _b: u32 };
@group(0) @binding(0) var<uniform> tp: TP;
@group(0) @binding(1) var<storage, read> ctl: array<u32>;
@group(0) @binding(2) var<storage, read> outIdx: array<u32>;
@group(0) @binding(3) var<storage, read> outVal: array<u32>;
@group(0) @binding(4) var<storage, read_write> tail: array<u32>;   // [cap idx | cap val bits]
@compute @workgroup_size(${WG})
fn main(@builtin(global_invocation_id) gid: vec3u) {
	let n = min(ctl[0] - tp.head, tp.cap);   // only dispatched when ctl[0] > head
	let i = gid.x;
	if (i >= n) { return; }
	tail[i] = outIdx[tp.head + i];
	tail[tp.cap + i] = outVal[tp.head + i];
}`;

export const K_HZ_CTL = defineKernel(
	"hz-ctl",
	HZ_CTL,
	[
		["tp", "uniform"],
		["starts", "read-only-storage"],
		["ctl", "storage"],
	],
	{ group: "look-haze" },
);
export const K_HZ_TAIL_PACK = defineKernel(
	"hz-tail-pack",
	HZ_TAIL_PACK,
	[
		["tp", "uniform"],
		["ctl", "read-only-storage"],
		["outIdx", "read-only-storage"],
		["outVal", "read-only-storage"],
		["tail", "storage"],
	],
	{ group: "look-haze" },
);

/**
 * Appends the GPU-controlled tail to a prep graph under construction (prepGraphFor's builder).
 * `capMax` is the compile-time window capacity (transient sizes are fixed at compile()).
 * Returns the handles the read node must include.
 */
export function addGpuTail<P extends TailParams>(
	g: ComputeGraph<P>,
	h: {
		tp: GraphBufferHandle; // uniform import, 16 B (head, cap)
		starts: GraphBufferHandle;
		outIdx: GraphBinding;
		outVal: GraphBinding;
	},
	capMax: number,
) {
	// ctl is written by a storage kernel and consumed as an indirect command: STORAGE | INDIRECT.
	const ctl = g.transientBuffer(
		"ctl",
		CTL_WORDS * 4,
		Buffer.STORAGE | Buffer.INDIRECT | Buffer.COPY_SRC | Buffer.COPY_DST,
	);
	const tail = g.transientBuffer("tail", 2 * capMax * 4);
	g.addKernel({
		id: "hz-ctl",
		spec: K_HZ_CTL,
		bindings: { tp: h.tp, starts: h.starts, ctl },
		workgroups: [1],
		// every ctl word is written: "full" (no clear needed)
	});
	// The gated node goes through the raw addComputePass: KernelNode.condition only admits CPU
	// conditions (core/graph.ts KernelCondition = Extract<…, {source: "cpu"}>). Gap G3.
	const run = (k: Kernel): GPUCommandGraphComputeExecutable<P> => ({
		encode: ({ computePass, getBuffer, parameters }) => {
			const buf = (x: GraphBinding) => getBuffer(x);
			// the graph's proxy turns this ONE dispatch() into dispatchIndirect(ctl, 8); x here only
			// satisfies checkWorkgroups (upper bound). Exactly one dispatch per GPU-conditional node.
			encodeDispatch(
				computePass,
				k,
				{
					tp: getBuffer(h.tp),
					ctl: getBuffer(ctl),
					outIdx: buf(h.outIdx),
					outVal: buf(h.outVal),
					tail: getBuffer(tail),
				},
				Math.max(1, Math.ceil(parameters.cap / WG)),
			);
		},
	});
	g.addComputePass({
		id: "tail-pack",
		condition: {
			id: "haze-head-overflow",
			source: "gpu",
			mode: "indirect",
			buffer: ctl,
			byteOffset: CTL_DISPATCH_BYTE,
		},
		resources: [
			{ buffer: h.tp, usage: "uniform" },
			{ buffer: ctl, usage: "storage-read" },
			{ buffer: h.outIdx, usage: "storage-read" },
			{ buffer: h.outVal, usage: "storage-read" },
			{ buffer: tail, usage: "storage-write" },
			// {buffer: ctl, usage: "indirect"} is added by the graph (addNode) when missing
		],
		workload: {
			operation: "hz-tail-pack",
			commandCount: 1,
			maximumWorkgroupCount: Math.ceil(capMax / WG),
			maximumInvocationCount: Math.ceil(capMax / WG) * WG,
			readByteLength: 2 * capMax * 4,
			writeByteLength: 2 * capMax * 4,
		},
		compile: ({ device }) => run(kernel(device, K_HZ_TAIL_PACK)),
		compileAsync: async ({ device }) => run(await kernelAsync(device, K_HZ_TAIL_PACK)),
	});
	// When the gate does not fire, `tail` is stale aliased bytes; the CPU reads only
	// [0, min(total-head, cap)) of it, decided by ctl[0] from the same read. So no clear is needed.
	return {
		ctl,
		tail,
		/** read-node ranges to append to prepGraphFor's g.readNode("head", [...]) */
		readRanges: [
			ctl,
			{ buffer: tail, size: (p: P) => p.cap * 4 },
			{ buffer: tail, offset: capMax * 4, size: (p: P) => p.cap * 4 },
		],
	};
}

/** CPU side after the single read: decides whether the residual exact tail read is still needed. */
export function afterRead(ctlBytes: ArrayBuffer, head: number, cap: number) {
	const ctl = new Uint32Array(ctlBytes);
	const total = ctl[0];
	const packed = total > head ? Math.min(total - head, cap) : 0;
	return {
		total,
		packed,
		/** bytes still to read exactly ([head + cap, total) of outIdx / outVal): rare residual */
		residual: Math.max(0, total - head - cap) * 4,
	};
}

// ---------------------------------------------------------------------------------------------
// The same predicate expressed through the upstream GPUProgram IR (the only public way to get an
// arena-backed GPUScalar: GPUScalar / createGPUScalar / GPUScalarDispatchGate / GPUValueArena are
// NOT exported from @luma.gl/gpgpu/gpu-core in alpha.2 or master 7d1d11e9).
// Limitation shown here: `head` must be a GPUProgramScalarLiteral, i.e. baked at compile time —
// there is no per-encoding parameter → scalar lowering, so the adaptive head cannot be an input.
// ---------------------------------------------------------------------------------------------
export function programOverflowDemo(device: Device, head: number) {
	const program = new GPUProgram({ id: "haze-tail-program" });
	const total = program.scalar("total", "uint32");
	const headS = program.scalar("head", "uint32");
	const overflow = program.scalar("overflow", "uint32");
	const marker = program.scalar("marker", "uint32");
	program.add([
		new GPUProgramScalarLiteral({ output: headS, value: head }),
		// `total` would be written by a custom node binding compilation.scalars.get("total").arena.buffer
		scalarCompare({ operation: "greater-than", left: total, right: headS, output: overflow }),
		new GPUConditionalOperation({
			id: "on-overflow",
			predicate: { id: "overflow", source: "gpu", value: overflow },
			// only semantic ops with registered lowerings (or primitives whose compute nodes carry
			// dispatchWorkgroups metadata) may sit under a runtime predicate
			body: new GPUProgramScalarLiteral({ output: marker, value: 1 }),
		}),
	]);
	const c: GPUProgramCompilation = new GPUProgramCompiler(device).compile(program);
	const totalScalar = c.scalars.get("total");
	if (!totalScalar) throw new Error("unreachable");
	// The arena is a graph TRANSIENT (never zeroed, uninitialised each encoding): a custom producer
	// must write word `totalScalar.wordOffset` of `totalScalar.arena.buffer` before compare runs, and
	// declare {buffer: totalScalar.arena.buffer, usage: "storage-read-write"} so the hazard order holds.
	const arena: GraphBufferHandle = totalScalar.arena.buffer;
	const word: number = totalScalar.wordOffset;
	return { graph: c.graph, arena, word, lowering: c.lowering, validation: c.validation };
}
